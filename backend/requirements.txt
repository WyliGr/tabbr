# Tabbr backend — Python dependencies
#
# SECURITY: every dependency is pinned to an exact version that has been
# verified against the OSV vulnerability database (https://api.osv.dev).
# Regenerate / re-verify the audit with:
#     python scripts/osv_audit.py backend/requirements.txt
#
# Prior to this file the backend used unpinned requirements, which means every
# container build silently pulled whatever "latest" the index served — a
# non-reproducible build that cannot be audited or that can break at any time
# (relevant after the homelab RCE/cryptominer incident).
#
# --- Direct dependencies -------------------------------------------------
fastapi==0.142.1
uvicorn[standard]==0.54.0
# The [asyncio] extra pulls in `greenlet`, which `sqlalchemy.ext.asyncio`
# requires. SQLAlchemy 2.1 no longer declares greenlet as a hard dependency,
# so without the extra the backend crashes on import with
# "The SQLAlchemy asyncio module requires that the Python 'greenlet' library
# is installed". This was already the case before the pins were added.
sqlalchemy[asyncio]==2.1.1
aiosqlite==0.22.1
asyncpg==0.31.0
pydantic==2.13.5
